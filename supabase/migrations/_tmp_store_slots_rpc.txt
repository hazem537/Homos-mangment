-- Store slots part 2: move movements to store_slot, tighten delivery check, rewrite RPCs.

-- Tighten destination check (slot required for store)
alter table public.warehouse_deliveries
  drop constraint if exists warehouse_deliveries_destination_check;

alter table public.warehouse_deliveries
  add constraint warehouse_deliveries_destination_check check (
    (
      destination_kind = 'warehouse'
      and warehouse_id is not null
      and section_id is not null
      and store_id is null
      and store_slot_id is null
    )
    or (
      destination_kind = 'store'
      and store_id is not null
      and store_slot_id is not null
      and warehouse_id is null
      and section_id is null
    )
  );

create or replace function public.check_delivery_section_warehouse()
returns trigger
language plpgsql
set search_path = public
as $$
declare
  v_warehouse_id uuid;
  v_store_id uuid;
begin
  if coalesce(new.destination_kind, 'warehouse') = 'store' then
    if new.store_id is null or new.store_slot_id is null then
      raise exception 'المخزن والخانة مطلوبان لتسليم المخزن';
    end if;

    select s.store_id into v_store_id
    from public.store_slots s
    where s.id = new.store_slot_id;

    if v_store_id is null then
      raise exception 'الخانة غير موجودة';
    end if;
    if v_store_id <> new.store_id then
      raise exception 'الخانة لا تتبع المخزن المحدد';
    end if;
    return new;
  end if;

  if new.section_id is null or new.warehouse_id is null then
    raise exception 'الشونة والربع مطلوبان لتسليم الشونة';
  end if;

  select s.warehouse_id into v_warehouse_id
  from public.warehouse_sections s
  where s.id = new.section_id;

  if v_warehouse_id is null then
    raise exception 'الربع غير موجود';
  end if;
  if v_warehouse_id <> new.warehouse_id then
    raise exception 'الربع لا يتبع الشونة المحددة';
  end if;
  return new;
end;
$$;

drop trigger if exists warehouse_deliveries_section_check on public.warehouse_deliveries;
create trigger warehouse_deliveries_section_check
  before insert or update of section_id, warehouse_id, store_id, store_slot_id, destination_kind
  on public.warehouse_deliveries
  for each row execute function public.check_delivery_section_warehouse();

-- Migrate movements store → store_slot (رئيسي)
update public.stock_movements m
set from_kind = 'store_slot'::public.stock_location_kind,
    from_id = s.id
from public.store_slots s
where m.from_kind::text = 'store'
  and s.store_id = m.from_id
  and s.is_active = true
  and lower(trim(s.name)) = 'رئيسي';

update public.stock_movements m
set to_kind = 'store_slot'::public.stock_location_kind,
    to_id = s.id
from public.store_slots s
where m.to_kind::text = 'store'
  and s.store_id = m.to_id
  and s.is_active = true
  and lower(trim(s.name)) = 'رئيسي';

update public.stock_movements m
set from_kind = 'store_slot'::public.stock_location_kind,
    from_id = (
      select s.id
      from public.store_slots s
      where s.store_id = m.from_id
        and s.is_active = true
      order by s.created_at
      limit 1
    )
where m.from_kind::text = 'store';

update public.stock_movements m
set to_kind = 'store_slot'::public.stock_location_kind,
    to_id = (
      select s.id
      from public.store_slots s
      where s.store_id = m.to_id
        and s.is_active = true
      order by s.created_at
      limit 1
    )
where m.to_kind::text = 'store';

create or replace function public.location_remaining_kg(
  p_kind public.stock_location_kind,
  p_id uuid,
  p_product_id uuid,
  p_status_id uuid,
  p_season_id uuid default null
)
returns numeric
language sql
stable
security invoker
set search_path = public
as $$
  with season as (
    select s.id
    from public.seasons s
    where s.id = coalesce(
      p_season_id,
      (select a.id from public.seasons a where a.is_active = true limit 1)
    )
  ),
  inbound as (
    select coalesce(sum(d.quantity_kg), 0)::numeric(14, 3) as kg
    from public.warehouse_deliveries d
    where d.product_id = p_product_id
      and d.status_id = p_status_id
      and d.season_id = (select id from season)
      and d.is_cancelled = false
      and (
        (p_kind = 'section' and d.section_id = p_id)
        or (p_kind = 'store_slot' and d.store_slot_id = p_id)
        or (
          p_kind = 'store'
          and d.destination_kind = 'store'
          and d.store_id = p_id
        )
      )
  ),
  moved_in as (
    select coalesce(sum(m.quantity_kg), 0)::numeric(14, 3) as kg
    from public.stock_movements m
    where m.is_cancelled = false
      and m.product_id = p_product_id
      and m.to_status_id = p_status_id
      and m.season_id = (select id from season)
      and (
        (m.to_kind = p_kind and m.to_id = p_id)
        or (
          p_kind = 'store'
          and m.to_kind = 'store_slot'
          and m.to_id in (
            select ss.id from public.store_slots ss
            where ss.store_id = p_id and ss.is_active = true
          )
        )
      )
  ),
  moved_out as (
    select coalesce(sum(m.quantity_kg), 0)::numeric(14, 3) as kg
    from public.stock_movements m
    where m.is_cancelled = false
      and m.product_id = p_product_id
      and m.status_id = p_status_id
      and m.season_id = (select id from season)
      and (
        (m.from_kind = p_kind and m.from_id = p_id)
        or (
          p_kind = 'store'
          and m.from_kind = 'store_slot'
          and m.from_id in (
            select ss.id from public.store_slots ss
            where ss.store_id = p_id and ss.is_active = true
          )
        )
      )
  )
  select (
    case
      when p_kind in ('section', 'store', 'store_slot') then coalesce((select kg from inbound), 0)
      else 0
    end
    + coalesce((select kg from moved_in), 0)
    - coalesce((select kg from moved_out), 0)
  )::numeric(14, 3);
$$;

grant execute on function public.location_remaining_kg(
  public.stock_location_kind, uuid, uuid, uuid, uuid
) to authenticated;

create or replace function public.store_balances(p_season_id uuid default null)
returns table (
  store_id uuid,
  store_name text,
  product_id uuid,
  product_name text,
  status_id uuid,
  status_name text,
  notes text,
  is_active boolean,
  kg_total numeric
)
language sql
stable
security invoker
set search_path = public
as $$
  with season as (
    select s.id
    from public.seasons s
    where s.id = coalesce(
      p_season_id,
      (select a.id from public.seasons a where a.is_active = true limit 1)
    )
  ),
  pairs as (
    select distinct ss.store_id, m.product_id, m.to_status_id as status_id
    from public.stock_movements m
    join public.store_slots ss on ss.id = m.to_id
    where m.to_kind = 'store_slot'
      and m.season_id = (select id from season)
      and m.is_cancelled = false
    union
    select distinct ss.store_id, m.product_id, m.status_id
    from public.stock_movements m
    join public.store_slots ss on ss.id = m.from_id
    where m.from_kind = 'store_slot'
      and m.season_id = (select id from season)
      and m.is_cancelled = false
    union
    select distinct d.store_id, d.product_id, d.status_id
    from public.warehouse_deliveries d
    where d.destination_kind = 'store'
      and d.store_id is not null
      and d.season_id = (select id from season)
      and d.is_cancelled = false
  )
  select
    st.id as store_id,
    st.name as store_name,
    pr.product_id,
    prod.name as product_name,
    pr.status_id,
    ss.name as status_name,
    st.notes,
    st.is_active,
    case
      when pr.product_id is null or pr.status_id is null then 0::numeric(14, 3)
      else public.location_remaining_kg(
        'store',
        st.id,
        pr.product_id,
        pr.status_id,
        (select id from season)
      )
    end as kg_total
  from public.stores st
  left join pairs pr on pr.store_id = st.id
  left join public.products prod on prod.id = pr.product_id
  left join public.stock_statuses ss on ss.id = pr.status_id
  where st.is_active = true
  order by st.name, prod.name nulls first, ss.name nulls first;
$$;

grant execute on function public.store_balances(uuid) to authenticated;

create or replace function public.store_slot_balances(
  p_store_id uuid,
  p_season_id uuid default null
)
returns table (
  slot_id uuid,
  slot_name text,
  slot_color text,
  product_id uuid,
  product_name text,
  status_id uuid,
  status_name text,
  kg_total numeric
)
language sql
stable
security invoker
set search_path = public
as $$
  with season as (
    select s.id
    from public.seasons s
    where s.id = coalesce(
      p_season_id,
      (select a.id from public.seasons a where a.is_active = true limit 1)
    )
  ),
  pairs as (
    select distinct m.to_id as slot_id, m.product_id, m.to_status_id as status_id
    from public.stock_movements m
    join public.store_slots ss on ss.id = m.to_id
    where m.to_kind = 'store_slot'
      and ss.store_id = p_store_id
      and m.season_id = (select id from season)
      and m.is_cancelled = false
    union
    select distinct m.from_id, m.product_id, m.status_id
    from public.stock_movements m
    join public.store_slots ss on ss.id = m.from_id
    where m.from_kind = 'store_slot'
      and ss.store_id = p_store_id
      and m.season_id = (select id from season)
      and m.is_cancelled = false
    union
    select distinct d.store_slot_id, d.product_id, d.status_id
    from public.warehouse_deliveries d
    where d.destination_kind = 'store'
      and d.store_id = p_store_id
      and d.store_slot_id is not null
      and d.season_id = (select id from season)
      and d.is_cancelled = false
  )
  select
    sl.id as slot_id,
    sl.name as slot_name,
    sl.color as slot_color,
    pr.product_id,
    prod.name as product_name,
    pr.status_id,
    st.name as status_name,
    case
      when pr.product_id is null or pr.status_id is null then 0::numeric(14, 3)
      else public.location_remaining_kg(
        'store_slot',
        sl.id,
        pr.product_id,
        pr.status_id,
        (select id from season)
      )
    end as kg_total
  from public.store_slots sl
  left join pairs pr on pr.slot_id = sl.id
  left join public.products prod on prod.id = pr.product_id
  left join public.stock_statuses st on st.id = pr.status_id
  where sl.store_id = p_store_id
    and sl.is_active = true
  order by sl.name, prod.name nulls first, st.name nulls first;
$$;

grant execute on function public.store_slot_balances(uuid, uuid) to authenticated;

create or replace function public.stock_location_label(
  p_kind public.stock_location_kind,
  p_id uuid,
  p_context_warehouse_id uuid default null
)
returns text
language sql
stable
security invoker
set search_path = public
as $$
  select case p_kind
    when 'waste' then 'هدر'
    when 'sale' then 'بيع'
    when 'section' then coalesce((
      select case
        when p_context_warehouse_id is not null
          and s.warehouse_id = p_context_warehouse_id then s.name
        when w.name is not null then 'شونة ' || w.name || ': ' || s.name
        else s.name
      end
      from public.warehouse_sections s
      left join public.warehouses w on w.id = s.warehouse_id
      where s.id = p_id
    ), 'ربع')
    when 'store_slot' then coalesce((
      select 'مخزن ' || coalesce(st.name, '—') || ': ' || ss.name
      from public.store_slots ss
      left join public.stores st on st.id = ss.store_id
      where ss.id = p_id
    ), 'خانة')
    when 'store' then 'مخزن: ' || coalesce((select name from public.stores where id = p_id), '—')
    when 'mekla' then 'مقلى: ' || coalesce((select name from public.meklas where id = p_id), '—')
    when 'client' then 'عميل: ' || coalesce((select name from public.clients where id = p_id), '—')
    else '—'
  end;
$$;

grant execute on function public.stock_location_label(
  public.stock_location_kind, uuid, uuid
) to authenticated;

declare
  def text;
begin
  def := pg_get_functiondef('public.supplier_detail_bundle(uuid,uuid,uuid)'::regprocedure);

  if position('store_slots' in def) = 0 then
    def := replace(
      def,
      $$'stores', coalesce((
        select jsonb_agg(jsonb_build_object('id', s.id, 'name', s.name) order by s.name)
        from public.stores s
        where s.is_active = true
      ), '[]'::jsonb),$$,
      $$'stores', coalesce((
        select jsonb_agg(jsonb_build_object('id', s.id, 'name', s.name) order by s.name)
        from public.stores s
        where s.is_active = true
      ), '[]'::jsonb),
      'store_slots', coalesce((
        select jsonb_agg(jsonb_build_object(
          'id', ss.id,
          'name', ss.name,
          'store_id', ss.store_id,
          'color', ss.color
        ) order by ss.name)
        from public.store_slots ss
        where ss.is_active = true
      ), '[]'::jsonb),$$
    );
  end if;

  if position('store_slot_id' in def) = 0 then
    def := replace(
      def,
      $$'store_id', d.store_id,$$,
      $$'store_id', d.store_id,
          'store_slot_id', d.store_slot_id,$$
    );
  end if;

  execute def;
end
$fix$;
